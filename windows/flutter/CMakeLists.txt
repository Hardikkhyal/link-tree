# This file controls Flutter-level build steps. It should not be edited.
cmake_minimum_required(VERSION 3.14)

set(EPHEMERAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ephemeral")

# Configuration provided via flutter tool.
# This file is written by `flutter build windows` before CMake runs and
# defines the real flutter_assemble custom command + FLUTTER_ROOT etc.
include(${EPHEMERAL_DIR}/generated_config.cmake OPTIONAL)

# Set fallback configurations for older versions of the flutter tool.
if (NOT DEFINED FLUTTER_TARGET_PLATFORM)
  set(FLUTTER_TARGET_PLATFORM "windows-x64")
endif()

# === Flutter Library ===
set(FLUTTER_LIBRARY "${EPHEMERAL_DIR}/flutter_windows.dll.lib")
set(FLUTTER_LIBRARY ${FLUTTER_LIBRARY} PARENT_SCOPE)
set(FLUTTER_ICU_DATA_FILE "${EPHEMERAL_DIR}/icudtl.dat" PARENT_SCOPE)
set(PROJECT_BUILD_DIR "${PROJECT_DIR}/build/" PARENT_SCOPE)
set(AOT_LIBRARY "${PROJECT_DIR}/build/windows/app.so" PARENT_SCOPE)

# The flutter INTERFACE target exposes the ephemeral include directories to
# all plugin and app targets that link against it.
add_library(flutter INTERFACE)
target_include_directories(flutter INTERFACE
  "${EPHEMERAL_DIR}"
  "${EPHEMERAL_DIR}/cpp_client_wrapper/include"
)
target_link_libraries(flutter INTERFACE "${FLUTTER_LIBRARY}")
add_dependencies(flutter flutter_assemble)

# === Wrapper libraries ===
# These source files are generated by flutter_assemble into the ephemeral dir.
# We declare them as GENERATED so CMake does not require them to exist at
# configure time — they will be created by the flutter_assemble build step.
set(WRAPPER_ROOT "${EPHEMERAL_DIR}/cpp_client_wrapper")

set(CPP_WRAPPER_SOURCES_CORE
  "${WRAPPER_ROOT}/core_implementations.cc"
  "${WRAPPER_ROOT}/standard_codec.cc"
)
set_source_files_properties(${CPP_WRAPPER_SOURCES_CORE} PROPERTIES GENERATED TRUE)

set(CPP_WRAPPER_SOURCES_PLUGIN
  "${WRAPPER_ROOT}/plugin_registrar.cc"
)
set_source_files_properties(${CPP_WRAPPER_SOURCES_PLUGIN} PROPERTIES GENERATED TRUE)

set(CPP_WRAPPER_SOURCES_APP
  "${WRAPPER_ROOT}/flutter_engine.cc"
  "${WRAPPER_ROOT}/flutter_view_controller.cc"
)
set_source_files_properties(${CPP_WRAPPER_SOURCES_APP} PROPERTIES GENERATED TRUE)

# Wrapper library for plugins.
add_library(flutter_wrapper_plugin STATIC
  ${CPP_WRAPPER_SOURCES_CORE}
  ${CPP_WRAPPER_SOURCES_PLUGIN}
)
apply_standard_settings(flutter_wrapper_plugin)
set_target_properties(flutter_wrapper_plugin PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden)
target_link_libraries(flutter_wrapper_plugin PUBLIC flutter)
target_include_directories(flutter_wrapper_plugin PUBLIC
  "${EPHEMERAL_DIR}"
  "${WRAPPER_ROOT}/include"
)
add_dependencies(flutter_wrapper_plugin flutter_assemble)

# Wrapper library for apps.
add_library(flutter_wrapper_app STATIC
  ${CPP_WRAPPER_SOURCES_CORE}
  ${CPP_WRAPPER_SOURCES_APP}
)
apply_standard_settings(flutter_wrapper_app)
target_link_libraries(flutter_wrapper_app PUBLIC flutter)
target_include_directories(flutter_wrapper_app PUBLIC
  "${EPHEMERAL_DIR}"
  "${WRAPPER_ROOT}/include"
)
add_dependencies(flutter_wrapper_app flutter_assemble)

# === flutter_assemble ===
# The real flutter_assemble is defined in generated_config.cmake (written by
# flutter build windows before CMake runs). It invokes tool_backend.bat which
# downloads the Flutter engine and extracts headers/DLLs into ephemeral/.
# This fallback no-op is only active on first-configure before that file exists.
if (NOT TARGET flutter_assemble)
  add_custom_target(flutter_assemble ALL
    COMMENT "flutter_assemble: Flutter tool backend will populate ephemeral/"
  )
endif()
